   241→   241→function ClientFormDialog({
   242→   242→  open,
   243→   243→  onOpenChange,
   244→   244→  client,
   245→   245→  onSaved,
   246→   246→}: {
   247→   247→  open: boolean;
   248→   248→  onOpenChange: (open: boolean) => void;
   249→   249→  client: Client | null;
   250→   250→  onSaved: () => void;
   251→   251→}) {
   252→   252→  const isEditing = !!client;
   253→   253→
   254→   254→  const form = useForm<ClientFormValues>({
   255→   255→    resolver: zodResolver(clientFormSchema),
   256→   256→    defaultValues: {
   257→   257→      firstName: client?.firstName ?? '',
   258→   258→      lastName: client?.lastName ?? '',
   259→   259→      phone: client?.phone ?? '',
   260→   260→      email: client?.email ?? '',
   261→   261→      address: client?.address ?? '',
   262→   262→      remarks: client?.remarks ?? '',
   263→   263→    },
   264→   264→  });
   265→   265→
   266→   266→  // Reset form when dialog opens with a different client
   267→   267→  useEffect(() => {
   268→   268→    if (open) {
   269→   269→      form.reset({
   270→   270→        firstName: client?.firstName ?? '',
   271→   271→        lastName: client?.lastName ?? '',
   272→   272→        phone: client?.phone ?? '',
   273→   273→        email: client?.email ?? '',
   274→   274→        address: client?.address ?? '',
   275→   275→        remarks: client?.remarks ?? '',
   276→   276→      });
   277→   277→    }
   278→   278→  }, [open, client, form]);
   279→   279→
   280→   280→  const [submitting, setSubmitting] = useState(false);
   281→   281→
   282→   282→  const onSubmit = async (values: ClientFormValues) => {
   283→   283→    setSubmitting(true);
   284→   284→    try {
   285→   285→      const payload: Record<string, string> = {
   286→   286→        firstName: values.firstName,
   287→   287→        lastName: values.lastName,
   288→   288→      };
   289→   289→      if (values.phone) payload.phone = values.phone;
   290→   290→      if (values.email) payload.email = values.email;
   291→   291→      if (values.address) payload.address = values.address;
   292→   292→      if (values.remarks) payload.remarks = values.remarks;
   293→   293→
   294→   294→      if (isEditing && client) {
   295→   295→        await fetch('/api/clients', {
   296→   296→          method: 'PUT',
   297→   297→          headers: getAuthHeaders(),
   298→   298→          body: JSON.stringify({ id: client.id, ...payload }),
   299→   299→        });
   300→   300→      } else {
   301→   301→        await fetch('/api/clients', {
   302→   302→          method: 'POST',
   303→   303→          headers: getAuthHeaders(),
   304→   304→          body: JSON.stringify(payload),
   305→   305→        });
   306→   306→      }
   307→   307→      onSaved();
   308→   308→      onOpenChange(false);
   309→   309→    } catch {
   310→   310→      // Error handled silently, could add toast
   311→   311→    } finally {
   312→   312→      setSubmitting(false);
   313→   313→    }
   314→   314→  };
   315→   315→
   316→   316→  return (
   317→   317→    <Dialog open={open} onOpenChange={onOpenChange}>
   318→   318→      <DialogContent className="sm:max-w-md">
   319→   319→        <DialogHeader>
   320→   320→          <DialogTitle>
   321→   321→            {isEditing ? 'Modifier le client' : 'Nouveau client'}
   322→   322→          </DialogTitle>
   323→   323→        </DialogHeader>
   324→   324→
   325→   325→        <Form {...form}>
   326→   326→          <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   327→   327→            <div className="grid grid-cols-2 gap-4">
   328→   328→              <FormField
   329→   329→                control={form.control}
   330→   330→                name="firstName"
   331→   331→                render={({ field }) => (
   332→   332→                  <FormItem>
   333→   333→                    <FormLabel>Prénom</FormLabel>
   334→   334→                    <FormControl>
   335→   335→                      <Input placeholder="Prénom" {...field} />
   336→   336→                    </FormControl>
   337→   337→                    <FormMessage />
   338→   338→                  </FormItem>
   339→   339→                )}
   340→   340→              />
   341→   341→              <FormField
   342→   342→                control={form.control}
   343→   343→                name="lastName"
   344→   344→                render={({ field }) => (
   345→   345→                  <FormItem>
   346→   346→                    <FormLabel>Nom</FormLabel>
   347→   347→                    <FormControl>
   348→   348→                      <Input placeholder="Nom" {...field} />
   349→   349→                    </FormControl>
   350→   350→                    <FormMessage />
   351→   351→                  </FormItem>
   352→   352→                )}
   353→   353→              />
   354→   354→            </div>
   355→   355→
   356→   356→            <FormField
   357→   357→              control={form.control}
   358→   358→              name="phone"
   359→   359→              render={({ field }) => (
   360→   360→                <FormItem>
   361→   361→                  <FormLabel>Téléphone</FormLabel>
   362→   362→                  <FormControl>
   363→   363→                    <Input placeholder="0X XX XX XX XX" {...field} />
   364→   364→                  </FormControl>
   365→   365→                  <FormMessage />
   366→   366→                </FormItem>
   367→   367→              )}
   368→   368→            />
   369→   369→
   370→   370→            <FormField
   371→   371→              control={form.control}
   372→   372→              name="email"
   373→   373→              render={({ field }) => (
   374→   374→                <FormItem>
   375→   375→                  <FormLabel>Email</FormLabel>
   376→   376→                  <FormControl>
   377→   377→                    <Input type="email" placeholder="email@exemple.com" {...field} />
   378→   378→                  </FormControl>
   379→   379→                  <FormMessage />
   380→   380→                </FormItem>
   381→   381→              )}
   382→   382→            />
   383→   383→
   384→   384→            <FormField
   385→   385→              control={form.control}
   386→   386→              name="address"
   387→   387→              render={({ field }) => (
   388→   388→                <FormItem>
   389→   389→                  <FormLabel>Adresse</FormLabel>
   390→   390→                  <FormControl>
   391→   391→                    <Input placeholder="Adresse" {...field} />
   392→   392→                  </FormControl>
   393→   393→                  <FormMessage />
   394→   394→                </FormItem>
   395→   395→              )}
   396→   396→            />
   397→   397→
   398→   398→            <FormField
   399→   399→              control={form.control}
   400→   400→              name="remarks"
   401→   401→              render={({ field }) => (
   402→   402→                <FormItem>
   403→   403→                  <FormLabel>Remarques</FormLabel>
   404→   404→                  <FormControl>
   405→   405→                    <Textarea
   406→   406→                      placeholder="Notes ou remarques..."
   407→   407→                      className="resize-none"
   408→   408→                      rows={3}
   409→   409→                      {...field}
   410→   410→                    />
   411→   411→                  </FormControl>
   412→   412→                  <FormMessage />
   413→   413→                </FormItem>
   414→   414→              )}
   415→   415→            />
   416→   416→
   417→   417→            <DialogFooter className="pt-2">
   418→   418→              <Button
   419→   419→                type="button"
   420→   420→                variant="outline"
   421→   421→                onClick={() => onOpenChange(false)}
   422→   422→                disabled={submitting}
   423→   423→              >
   424→   424→                Annuler
   425→   425→              </Button>
   426→   426→              <Button type="submit" disabled={submitting}>
   427→   427→                {submitting ? (
   428→   428→                  <span className="flex items-center gap-2">
   429→   429→                    <span className="h-4 w-4 animate-spin rounded-full border-2 border-current border-t-transparent" />
   430→   430→                    {isEditing ? 'Enregistrement...' : 'Création...'}

   433→   433→                  'Enregistrer'

   435→   435→                  'Créer le client'
   436→   436→                )}
   437→   437→              </Button>

   448→   448→function ClientHistoryDialog({
   449→   449→  open,
   450→   450→  onOpenChange,
   451→   451→  client,
   452→   452→}: {
   453→   453→  open: boolean;
   454→   454→  onOpenChange: (open: boolean) => void;
   455→   455→  client: Client | null;
   456→   456→}) {
   457→   457→  const [appointments, setAppointments] = useState<Appointment[]>([]);
   458→   458→  const [invoices, setInvoices] = useState<Invoice[]>([]);
   459→   459→  const [loadingAppointments, setLoadingAppointments] = useState(true);
   460→   460→  const [loadingInvoices, setLoadingInvoices] = useState(true);
   461→   461→
   462→   462→  useEffect(() => {
   463→   463→    if (open && client) {
   464→   464→      fetch(`/api/clients/${client.id}/appointments`, { headers: getAuthHeaders() })
   465→   465→        .then((res) => res.json())
   466→   466→        .then((data) => setAppointments(Array.isArray(data) ? data : []))
   467→   467→        .catch(() => setAppointments([]))
   468→   468→        .finally(() => setLoadingAppointments(false));
   469→   469→
   470→   470→      fetch(`/api/clients/${client.id}/invoices`, { headers: getAuthHeaders() })
   471→   471→        .then((res) => res.json())
   472→   472→        .then((data) => setInvoices(Array.isArray(data) ? data : []))
   473→   473→        .catch(() => setInvoices([]))
   474→   474→        .finally(() => setLoadingInvoices(false));
   475→   475→    }
   476→   476→  }, [open, client]);
   477→   477→
   478→   478→  if (!client) return null;
   479→   479→
   480→   480→  return (
   481→   481→    <Dialog open={open} onOpenChange={onOpenChange}>
   482→   482→      <DialogContent className="sm:max-w-2xl">
   483→   483→        <DialogHeader>
   484→   484→          <DialogTitle>
   485→   485→            Historique — {client.firstName} {client.lastName}
   486→   486→          </DialogTitle>
   487→   487→        </DialogHeader>
   488→   488→
   489→   489→        <Tabs defaultValue="appointments" className="w-full">
   490→   490→          <TabsList className="w-full">
   491→   491→            <TabsTrigger value="appointments" className="flex-1">
   492→   492→              Rendez-vous
   493→   493→            </TabsTrigger>
   494→   494→            <TabsTrigger value="invoices" className="flex-1">
   495→   495→              Factures
   496→   496→            </TabsTrigger>
   497→   497→          </TabsList>
   498→   498→
   499→   499→          <TabsContent value="appointments">
   500→   500→            <ScrollArea className="max-h-96 overflow-y-auto">
   501→   501→              {loadingAppointments ? (
   502→   502→                <HistorySkeleton />
   503→   503→              ) : appointments.length === 0 ? (
   504→   504→                <div className="text-center py-8 text-muted-foreground text-sm">
   505→   505→                  Aucun rendez-vous trouvé
   506→   506→                </div>
   507→   507→              ) : (
   508→   508→                <Table>
   509→   509→                  <TableHeader>
   510→   510→                    <TableRow>
   511→   511→                      <TableHead>Date</TableHead>
   512→   512→                      <TableHead>Service</TableHead>
   513→   513→                      <TableHead>Employé</TableHead>
   514→   514→                      <TableHead>Statut</TableHead>
   515→   515→                    </TableRow>
   516→   516→                  </TableHeader>
   517→   517→                  <TableBody>
   518→   518→                    {appointments.map((appt) => (

   802→   802→export function ClientsModule() {
   803→   803→  const [clients, setClients] = useState<Client[]>([]);
   804→   804→  const [loading, setLoading] = useState(true);
   805→   805→  const [searchQuery, setSearchQuery] = useState('');
   806→   806→  const [debouncedQuery, setDebouncedQuery] = useState('');
   807→   807→
   808→   808→  // Dialogs
   809→   809→  const [formDialogOpen, setFormDialogOpen] = useState(false);
   810→   810→  const [editingClient, setEditingClient] = useState<Client | null>(null);
   811→   811→
   812→   812→  const [historyDialogOpen, setHistoryDialogOpen] = useState(false);
   813→   813→  const [historyClient, setHistoryClient] = useState<Client | null>(null);
   814→   814→
   815→   815→  const [deletingClient, setDeletingClient] = useState<Client | null>(null);
   816→   816→  const [deleting, setDeleting] = useState(false);
   817→   817→
   818→   818→  // ── Debounced search ──
   819→   819→  useEffect(() => {
   820→   820→    const timer = setTimeout(() => {
   821→   821→      setDebouncedQuery(searchQuery);
   822→   822→    }, 300);
   823→   823→    return () => clearTimeout(timer);
   824→   824→  }, [searchQuery]);
   825→   825→
   826→   826→  // ── Fetch clients ──
   827→   827→  const fetchClients = useCallback(async () => {
   828→   828→    setLoading(true);
   829→   829→    try {
   830→   830→      const params = debouncedQuery ? `?q=${encodeURIComponent(debouncedQuery)}` : '';
   831→   831→      const res = await fetch(`/api/clients${params}`, {
   832→   832→        headers: getAuthHeaders(),
   833→   833→      });
   834→   834→      const data = await res.json();
   835→   835→      setClients(Array.isArray(data) ? data : []);
   836→   836→    } catch {
   837→   837→      setClients([]);
   838→   838→    } finally {
   839→   839→      setLoading(false);
   840→   840→    }
   841→   841→  }, [debouncedQuery]);
   842→   842→
   843→   843→  useEffect(() => {
   844→   844→    fetchClients();
   845→   845→  }, [fetchClients]);
   846→   846→
   847→   847→  // ── Handlers ──
   848→   848→  const handleCreateNew = () => {
   849→   849→    setEditingClient(null);
   850→   850→    setFormDialogOpen(true);
   851→   851→  };
   852→   852→
   853→   853→  const handleEdit = (client: Client) => {
   854→   854→    setEditingClient(client);
   855→   855→    setFormDialogOpen(true);
   856→   856→  };
   857→   857→
   858→   858→  const handleSaved = () => {
   859→   859→    fetchClients();
   860→   860→  };
   861→   861→
   862→   862→  const handleHistory = (client: Client) => {
   863→   863→    setHistoryClient(client);
   864→   864→    setHistoryDialogOpen(true);
   865→   865→  };
   866→   866→
   867→   867→  const handleDelete = async () => {
   868→   868→    if (!deletingClient) return;
   869→   869→    setDeleting(true);
   870→   870→    try {
   871→   871→      await fetch('/api/clients', {
   872→   872→        method: 'DELETE',
   873→   873→        headers: getAuthHeaders(),
   874→   874→        body: JSON.stringify({ id: deletingClient.id }),
   875→   875→      });
   876→   876→      setClients((prev) => prev.filter((c) => c.id !== deletingClient.id));
   877→   877→    } catch {
   878→   878→      // Error handled silently
   879→   879→    } finally {
   880→   880→      setDeleting(false);
   881→   881→      setDeletingClient(null);
   882→   882→    }
   883→   883→  };
   884→   884→
   885→   885→  return (
   886→   886→    <div className="space-y-6">
   887→   887→      {/* ── Header ── */}
   888→   888→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   889→   889→        <div>
   890→   890→          <h1 className="text-2xl font-bold tracking-tight">
   891→   891→            Gestion des clients
   892→   892→          </h1>
   893→   893→          <p className="text-sm text-muted-foreground mt-1">
   894→   894→            {clients.length} client{clients.length !== 1 ? 's' : ''} au total
   895→   895→          </p>
   896→   896→        </div>
   897→   897→        <Button onClick={handleCreateNew} className="w-full sm:w-auto">
   898→   898→          <Plus className="h-4 w-4 mr-2" />
   899→   899→          Nouveau client
   900→   900→        </Button>
   901→   901→      </div>
   902→   902→
   903→   903→      {/* ── Search Bar ── */}
   904→   904→      <div className="relative max-w-md">
   905→   905→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   906→   906→        <Input
   907→   907→          placeholder="Rechercher par nom ou téléphone..."
   908→   908→          value={searchQuery}
   909→   909→          onChange={(e) => setSearchQuery(e.target.value)}
   910→   910→          className="pl-9"
   911→   911→        />
   912→   912→      </div>
   913→   913→
   914→   914→      {/* ── Content ── */}
   915→   915→      {loading ? (
   916→   916→        <>
   917→   917→          {/* Desktop skeleton */}
   918→   918→          <div className="hidden md:block">
   919→   919→            <Card>
   920→   920→              <CardContent className="p-4">
   921→   921→                <TableSkeleton />
   922→   922→              </CardContent>
   923→   923→            </Card>
   924→   924→          </div>
   925→   925→          {/* Mobile skeleton */}
   926→   926→          <div className="md:hidden">
   927→   927→            <CardSkeleton />
   928→   928→          </div>
   929→   929→        </>
   930→   930→      ) : (
   931→   931→        <>

   933→   933→          <div className="hidden md:block">
   934→   934→            <ClientTable
   935→   935→              clients={clients}
   936→   936→              onEdit={handleEdit}
   937→   937→              onDelete={setDeletingClient}
   938→   938→              onHistory={handleHistory}
   939→   939→            />
   940→   940→          </div>
   941→   941→          {/* Mobile cards (hidden on desktop) */}
   942→   942→          <div className="md:hidden">
   943→   943→            <ClientCardList
   944→   944→              clients={clients}
   945→   945→              onEdit={handleEdit}
   946→   946→              onDelete={setDeletingClient}
   947→   947→              onHistory={handleHistory}
   948→   948→            />
   949→   949→          </div>
   950→   950→        </>

   952→   952→
   953→   953→      {/* ── Create / Edit Dialog ── */}
   954→   954→      <ClientFormDialog
   955→   955→        open={formDialogOpen}

   957→   957→        client={editingClient}
   958→   958→        onSaved={handleSaved}
   959→   959→      />
   960→   960→
   961→   961→      {/* ── History Dialog ── */}
   962→   962→      <ClientHistoryDialog
   963→   963→        key={historyClient?.id}
   964→   964→        open={historyDialogOpen}
   965→   965→        onOpenChange={setHistoryDialogOpen}
   966→   966→        client={historyClient}
   967→   967→      />
   968→   968→
   969→   969→      {/* ── Delete Confirmation ── */}
   970→   970→      <AlertDialog
   971→   971→        open={!!deletingClient}
   972→   972→        onOpenChange={(open) => {
   973→   973→          if (!open) setDeletingClient(null);
   974→   974→        }}
   975→   975→      >
   976→   976→        <AlertDialogContent>
   977→   977→          <AlertDialogHeader>
   978→   978→            <AlertDialogTitle>Supprimer ce client ?</AlertDialogTitle>

   980→   980→              Cette action est irréversible. Le client{' '}

   982→   982→                {deletingClient?.firstName} {deletingClient?.lastName}
