    96→    96→  return new Date(dateStr).toLocaleDateString('fr-FR', {
    97→    97→    day: 'numeric',
    98→    98→    month: 'long',
    99→    99→    year: 'numeric',
   100→   100→    hour: '2-digit',
   101→   101→    minute: '2-digit',

   117→   117→function formatDuration(mins: number): string {
   118→   118→  if (mins < 60) return `${mins} min`;
   119→   119→  const h = Math.floor(mins / 60);
   120→   120→  const m = mins % 60;
   121→   121→  return m > 0 ? `${h}h ${m}min` : `${h}h`;
   122→   122→}

   154→   154→function statusDotColor(statuses: AppointmentStatus[]): string {
   155→   155→  if (statuses.length === 0) return 'transparent';
   156→   156→  const unique = [...new Set(statuses)];
   157→   157→  if (unique.length === 1) {
   158→   158→    if (unique[0] === 'PROGRAMME') return '#3b82f6';
   159→   159→    if (unique[0] === 'ANNULE') return '#ef4444';
   160→   160→    if (unique[0] === 'TERMINE') return '#10b981';
   161→   161→  }
   162→   162→  if (statuses.includes('PROGRAMME')) return '#3b82f6';
   163→   163→  if (statuses.includes('TERMINE')) return '#10b981';
   164→   164→  return '#ef4444';
   165→   165→}
